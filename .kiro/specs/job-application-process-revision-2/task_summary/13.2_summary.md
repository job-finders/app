### Validation System
```javascript
async validate_application(application_id, user_id) {
    // Check cover letter
    if (!application.cover_letter) {
        issues.push({
            "title": "Cover Letter Missing",
            "description": "Adding a cover letter significantly improves your chances...",
            "action_url": "/jobs/{job_id}#cover-letter",
            "action_text": "Add Cover Letter"
        });
    }
    
    // Check questionnaire completion
    if (job.required_questionnaire && !application.questionnaires_completed) {
        issues.push({
            "title": "Questionnaire Incomplete",
            "description": "This job requires completion of a questionnaire...",
            "action_url": "/api/applications/workflow/applications/{id}/questionnaires",
            "action_text": "Complete Questionnaire"
        });
    }
}
```

### Submission Process
```javascript
async handleFinalSubmission() {
    // Validate application before submission
    const validation_result = await this.validate_application(application_id, user_id);
    if (validation_result.issues.length > 0) {
        return { success: false, message: "Please address all validation issues..." };
    }
    
    // Submit application
    application.workflow_step = 'submitted';
    application.workflow_completed_at = datetime.utcnow();
    application.application_stage = 'SUBMITTED';
}
```
